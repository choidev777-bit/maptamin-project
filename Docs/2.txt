// Pseudo-code for scrapeNaverBatch (in scraper.ts)

// 1. 브라우저는 딱 한 번만 켭니다 (CPU/메모리 절약의 핵심)
// headless: true로 설정하여 서버 부하를 최소화합니다.
const globalBrowser = await chromium.launch({ headless: true });

try {
    for (const task of tasks) {
        let context = null;

        try {
            // 2. 매 작업마다 '새로운 격리 공간(Context)'을 만듭니다.
            // 이때 프록시 세션 ID를 랜덤으로 생성하여 새 IP를 할당받습니다.
            context = await globalBrowser.newContext({
                proxy: {
                    server: 'http://brd.superproxy.io:33335', // Bright Data 슈퍼 프록시 주소
                    // ⭐️ 핵심: 세션 ID를 매번 랜덤하게 생성 -> IP 교체 트리거
                    username: `brd-customer-{ID}-zone-{ZONE}-session-${Math.random().toString(36).substring(7)}`, 
                    password: '{PASSWORD}'
                },
                // 봇 탐지 회피를 위해 User-Agent도 랜덤화 추천 (선택 사항)
                userAgent: 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)...' 
            });

            // 3. 페이지 생성
            const page = await context.newPage();

            // 4. 리소스 차단 적용 (1단계 코드 함수 호출)
            // 반드시 페이지 생성 직후, 네이버 접속 전에 적용해야 합니다.
            await applyResourceBlocking(page);

            // 5. 검색 수행 (수정된 함수 호출)
            // 주의: scrapeAtLocation 함수는 더 이상 내부에서 브라우저를 켜지 않고,
            // 우리가 만든 'page'를 받아서 쓰도록 수정되어 있어야 합니다.
            const result = await scrapeAtLocation(page, task);
            
            // 결과 저장 로직...
            results.push(result);

        } catch (taskError) {
            console.error(`❌ Task failed for keyword "${task.keyword}":`, taskError);
            // 에러가 나더라도 다음 작업(Loop)은 계속 진행되어야 합니다.
        } finally {
            // 6. [매우 중요] 작업이 성공하든 실패하든 Context는 무조건 닫아야 합니다.
            // 이걸 안 하면 메모리가 계속 쌓여서 서버가 멈춥니다.
            if (context) await context.close();
        }
        
        // (선택) 너무 빠른 반복을 방지하기 위한 짧은 대기
        // await delay(500); 
    }

} finally {
    // 7. 모든 작업(Loop)이 끝나면 브라우저를 종료합니다.
    await globalBrowser.close();
}
