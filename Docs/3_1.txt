// Pseudo-code inside scrapeAtLocation (Refined)

let jsonFound = false;

// 1. 감시자 등록 (반드시 goto 전에!)
page.on('response', async (response) => {
    // 이미 찾았으면 더 이상 처리 안 함 (중복 방지)
    if (jsonFound) return; 

    const url = response.url();
    
    // 네이버 지도 API 패턴 매칭 (graphql, search 등)
    // 팁: url 패턴은 네이버 업데이트에 따라 바뀔 수 있으니 config로 빼두면 좋습니다.
    if (url.includes('/api/graphql') || (url.includes('search.naver') && url.includes('type=PLACE'))) {
        try {
            const data = await response.json();
            
            // 데이터 검증 (껍데기 JSON 제외)
            if (isValidData(data)) {
                console.log("🎯 HIT! JS 데이터 탈취 성공 (Fast Kill)");
                jsonFound = true;
                
                // ⭐️ 중요: 저장 로직이 비동기라면 반드시 await
                await saveResult(data); 
                
                // ⭐️ 핵심: 브라우저 강제 종료 (일부러 에러 발생시킴)
                await page.close(); 
            }
        } catch (err) {
            // JSON 파싱 에러 등은 무시하고 넘어감 (DOM 파싱 기회 제공)
        }
    }
});

// 2. 페이지 접속 & 하이브리드 실행
try {
    // waitUntil: 'domcontentloaded'로 설정하여 
    // 불필요한 이미지 로딩까지 기다리지 않게 함 (더 빠른 진입)
    await page.goto(`https://map.naver.com/p/search/${keyword}`, { 
        waitUntil: 'domcontentloaded',
        timeout: 10000 // 10초 안에 쇼부 못 보면 실패 처리
    });
    
    // 3. 페이지 로딩이 끝날 때까지 JSON을 못 찾았다면? -> Plan B 가동
    if (!jsonFound) {
        console.warn("⚠️ API 인터셉트 실패. DOM 스크래핑으로 전환 (Fallback)");
        
        // DOM 방식은 화면이 다 그려져야 하므로 선택자가 뜰 때까지 기다림
        await page.waitForSelector('span.TYaxT', { timeout: 5000 });
        await scrapeDOM(page);
    }

} catch (e: any) {
    // 4. Fast Kill로 인한 '의도된 에러'인지 확인
    // Playwright에서 page.close() 시 발생하는 에러 메시지 체크
    if (jsonFound && (e.message.includes('Target closed') || e.message.includes('closed'))) {
        // 이건 에러가 아니라 '성공'입니다.
        return { success: true, method: 'json', ... };
    }
    
    // 진짜 에러라면 상위로 던짐
    throw e;
}
